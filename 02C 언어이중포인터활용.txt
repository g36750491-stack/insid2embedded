1. 이중 포인터를 이용한 Call by Reference 
예제함수 내부에서 바깥의 포인터 변수 값 자체를 바꾸고 싶을 때 이중 포인터를 인자로 전달합니다. 
일반 포인터만으로는 포인터가 가리키는 변수의 값은 바꿀 수 있지만, 
포인터 변수 그 자체의 주소를 바꿀 수는 없기 때문입니다.
#include <stdio.h>

// 이중 포인터를 매개변수로 받아 포인터의 주소를 변경
void changePointer(int **ptr_addr, int *new_target) {
    *ptr_addr = new_target; // main 함수의 p 변수가 가리키는 대상 변경
}

int main(void) {
    int a = 10;
    int b = 20;
    int *p = &a; // p는 a를 가리킴

    printf("변경 전 *p: %d\n", *p); // 10

    // p 변수의 주소(&p)를 전달하므로 이중 포인터(int **)로 받음
    changePointer(&p, &b);

    printf("변경 후 *p: %d\n", *p); // 20 (p가 b를 가리키도록 변경됨)
    return 0;
}

2. 포인터 배열과의 관계포인터 배열의 정의: 
포인터들의 주소를 저장하는 배열입니다 (예: char *arr[3];).
이중 포인터와의 호환성: 포인터 배열의 이름은 배열의 첫 번째 요소의 주소, 즉 포인터의 주소(포인터*)를 가리키는 주소가 되므로
이중 포인터(**)로 안전하게 전달할 수 있습니다.
활용: 문자열 여러 개(문자열 배열)를 함수의 인자로 넘길 때 char ** 형태의 이중 포인터를 자주 사용합니다

#include <stdio.h>

// 포인터 배열을 이중 포인터로 받아 순회 출력
void printStrings(char **arr, int size) {
    for (int i = 0; i < size; i++) {
        printf("%s\n", arr[i]); // *(arr + i)와 동일
    }
}

int main(void) {
    // 문자열(char*)을 저장하는 포인터 배열
    char *fruits[3] = {"Apple", "Banana", "Cherry"};

    // fruits는 char* 배열의 시작 주소이므로 char** 로 전달 가능
    printStrings(fruits, 3);
    
    return 0;
}
